    4

    i.

    #include <stdio.h>
main()
{
  float x = 1.1;
  while (x == 1.1)
  {
    printf("\n%f", x);
    x = x - 0.1;
  }
}

    ii.

    #include <stdio.h>
main()
{
  int i = 4, j = -1, k = 0, y, z;
  y = i + 5 && j + 1 || k + 2;
  z = i + 5 || j + 1 && k + 2;
  printf("\ny = %d z = %d", y, z);
}

    iii.

    #include <stdio.h>
main()
{
  float a = 3.5;
  switch (a)
  {
    case 0.5:
      printf("Art of C");
      break;
    case 1.5:
      printf("Spirit of C");
      break;
    case 2.5:
      printf("See C");
      break;
    case 3.5:
      printf("Simply C");
  }
}

    iv.

    #include <stdio.h>
main()
{
  int x = 6, y = 3, z;
  while (x >= 0)
  {
    if (x == y)
      break;
    else
      printf("\n%d %d", x, y);
    x--;
    y++;
  }
}
